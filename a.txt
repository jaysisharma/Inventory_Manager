    // Create a new jsPDF instance
    const doc = new jsPDF();

    // Add title
    doc.setFontSize(18);
    doc.text("Product Details", 10, 20);

    // Add content
    doc.setFontSize(12);
    doc.text(`Product Name: ${formData.productName || "N/A"}`, 10, 40);
    doc.text(`Price: ${formData.price || "N/A"}`, 10, 50);
    doc.text(`Remaining Amount: ${formData.remainingAmount || "N/A"}`, 10, 60);
    doc.text(`Delivery Date: ${formData.deliveryDate || "N/A"}`, 10, 70);

    // Save PDF to a blob
    const pdfBlob = doc.output("blob");

    // Check if the Web Share API is supported
    if (navigator.share && navigator.canShare({ files: [pdfBlob] })) {
      try {
        // Convert the blob to a file
        const pdfFile = new File([pdfBlob], `${formData.customerName.pdf}`, {
          type: "application/pdf",
        });

        // Use Web Share API to share the file
        await navigator.share({
          title: "Product Details",
          text: "Check out this product details PDF",
          files: [pdfFile],
        });
        // alert("PDF shared successfully!");
      } catch (error) {
        // alert("Failed to share the PDF.");
      }
    } else {
      // Fallback: Trigger download if sharing is not supported
      doc.save(`${formData.customerName}.pdf`);
    }
  };
